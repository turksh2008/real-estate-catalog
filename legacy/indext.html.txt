  <script>
    var PLACEHOLDER_IMG = 'data:image/svg+xml;charset=UTF-8,' + encodeURIComponent(
      '<svg xmlns="http://www.w3.org/2000/svg" width="400" height="230">' +
      '<rect width="400" height="230" fill="#e2e8f0"/>' +
      '<text x="50%" y="50%" fill="#94a3b8" font-family="sans-serif" font-size="18" ' +
      'text-anchor="middle" dominant-baseline="middle">لا توجد صورة</text></svg>'
    );

    new Vue({
      el: '#app',
      data: {
        webAppUrl: 'https://script.google.com/macros/s/AKfycbz0QPRz1P_Q_pNHLS3fSlcLg_SvWAbWpIXLYnvgTq2bHAUkA0r0ImVwbb8vpVUXRlI8CQ/exec',
        properties: [],
        activeItem: null,
        loading: true,
        error: null,
        selectedStatus: 'all',
        selectedType: 'all'
      },
      computed: {
        filteredProperties: function() {
          var self = this;
          return this.properties.filter(function(item) {
            var isSoldItem = self.isSold(item);

            if (self.selectedStatus === 'available' && isSoldItem) return false;
            if (self.selectedStatus === 'sold' && !isSoldItem) return false;

            var listingType = String(item.listingType || '').toLowerCase();
            if (self.selectedType === 'sale' && !(listingType.indexOf('sale') !== -1 || listingType.indexOf('بيع') !== -1)) return false;
            if (self.selectedType === 'rent' && !(listingType.indexOf('rent') !== -1 || listingType.indexOf('إيجار') !== -1 || listingType.indexOf('ايجار') !== -1)) return false;

            return true;
          });
        }
      },
      methods: {
        /* ----- Data loading from Google Apps Script ----- */
        fetchProperties: function(attempt) {
          var self = this;
          attempt = attempt || 1;
          var MAX_ATTEMPTS = 3;
          var TIMEOUT_MS = 15000;

          this.loading = true;
          this.error = null;

          var controller = (typeof AbortController !== 'undefined') ? new AbortController() : null;
          var timer = setTimeout(function() { if (controller) controller.abort(); }, TIMEOUT_MS);

          var options = { method: 'GET', redirect: 'follow', cache: 'no-store' };
          if (controller) options.signal = controller.signal;

          fetch(this.webAppUrl + (this.webAppUrl.indexOf('?') === -1 ? '?' : '&') + 't=' + Date.now(), options)
            .then(function(res) {
              clearTimeout(timer);
              if (!res.ok) throw new Error('HTTP ' + res.status);
              return res.json();
            })
            .then(function(data) {
              var list = self.extractList(data);
              self.properties = list.map(function(row, i) { return self.normalizeProperty(row, i); });
              self.loading = false;
            })
            .catch(function(err) {
              clearTimeout(timer);
              console.error('Error fetching properties (attempt ' + attempt + '):', err);

              if (attempt < MAX_ATTEMPTS) {
                setTimeout(function() { self.fetchProperties(attempt + 1); }, 1000 * attempt);
                return;
              }

              self.loading = false;
              self.error = (err && err.name === 'AbortError')
                ? 'انتهت مهلة الاتصال بالخادم. تحقق من اتصالك بالإنترنت وحاول مجدداً.'
                : 'تعذّر تحميل العقارات حالياً. يرجى المحاولة مرة أخرى.';
            });
        },

        /* Accept a bare array or a wrapped payload ({ data: [...] }, etc.) */
        extractList: function(data) {
          if (Array.isArray(data)) return data;
          if (!data || typeof data !== 'object') return [];
          var keys = ['data', 'properties', 'items', 'records', 'rows', 'result'];
          for (var i = 0; i < keys.length; i++) {
            if (Array.isArray(data[keys[i]])) return data[keys[i]];
          }
          return [];
        },

        /* Map a sheet row (any casing) to a stable internal shape */
        normalizeProperty: function(p, index) {
          p = p || {};
          function pick() {
            for (var i = 0; i < arguments.length; i++) {
              var v = arguments[i];
              if (v !== undefined && v !== null && String(v).trim() !== '') return v;
            }
            return '';
          }
          var id = pick(p.id, p.ID, p.Id, p.rowId, p.RowId, p.reference, p.Reference);
          return {
            _key: (id || 'row') + '-' + index,
            id: id || ('row-' + index),
            title: pick(p.title, p.Title),
            location: pick(p.location, p.Location),
            price: pick(p.price, p.Price),
            listingType: pick(p.listingType, p.ListingType, p.type, p.Type),
            propertyType: pick(p.propertyType, p.PropertyType, p.category, p.Category),
            status: pick(p.status, p.Status, p.availability, p.Availability),
            bedrooms: pick(p.bedrooms, p.Bedrooms, p.beds, p.Beds),
            bathrooms: pick(p.bathrooms, p.Bathrooms, p.baths, p.Baths),
            landSize: pick(p.landSize, p.LandSize, p.area, p.Area, p.size, p.Size),
            description: pick(p.description, p.Description, p.details, p.Details),
            images: this.parseImages(pick(p.images, p.Images, p.image, p.Image, p.photos, p.Photos, p.pictures, p.Pictures)),
            activeImgIndex: 0
          };
        },

        /* Sheet cell may hold an array or a delimited string of URLs */
        parseImages: function(val) {
          var arr = [];
          if (Array.isArray(val)) {
            arr = val.slice();
          } else if (val !== undefined && val !== null && String(val).trim() !== '') {
            arr = String(val).split(/[\n,|;]+/);
          }
          arr = arr.map(function(s) { return String(s).trim(); })
                   .filter(function(s) { return s !== ''; });
          return arr.length ? arr : [PLACEHOLDER_IMG];
        },

        onImgError: function(event) {
          if (event && event.target && event.target.src !== PLACEHOLDER_IMG) {
            event.target.src = PLACEHOLDER_IMG;
          }
        },

        formatPrice: function(val) {
          if (val === null || val === undefined || String(val).trim() === '') return '—';
          var num = Number(String(val).replace(/[^0-9.\-]/g, ''));
          return isNaN(num) ? String(val) : num.toLocaleString('en-US');
        },

        openModal: function(item) {
          this.activeItem = item;
        },

        /* ----- Status helpers ----- */
        isSold: function(item) {
          if (!item) return false;
          var val = String(item.status || '').trim().toLowerCase();
          if (val === '') return false;
          return ['available', 'active', 'متاح', 'متوفر', 'نشط'].indexOf(val) === -1;
        },

        /* Arabic status label used inside the card / modal */
        getStatusText: function(item) {
          var val = String((item && item.status) || '').trim().toLowerCase();
          if (val === '' || val === 'available' || val === 'متاح') return 'متاح';
          if (val.indexOf('rent') !== -1 || val.indexOf('مؤجر') !== -1 || val.indexOf('إيجار') !== -1 || val.indexOf('ايجار') !== -1) return 'تم التأجير';
          if (val.indexOf('reserv') !== -1 || val.indexOf('محجوز') !== -1 || val.indexOf('حجز') !== -1) return 'محجوز';
          return 'تم البيع';
        },

        /* Short uppercase badge shown on the grayscale corner ribbon */
        getStatusBadge: function(item) {
          var val = String((item && item.status) || '').trim().toLowerCase();
          if (val.indexOf('rent') !== -1 || val.indexOf('مؤجر') !== -1 || val.indexOf('إيجار') !== -1 || val.indexOf('ايجار') !== -1) return 'RENTED';
          if (val.indexOf('reserv') !== -1 || val.indexOf('محجوز') !== -1 || val.indexOf('حجز') !== -1) return 'RESERVED';
          return 'SOLD';
        },

        scrollCarousel: function(item, direction) {
          var refs = this.$refs['carousel_' + item._key];
          if (!refs || !refs[0]) return;
          var container = refs[0];
          var slideWidth = container.clientWidth;

          // Browser-agnostic RTL scroll offset direction
          var offset = direction === 'next' ? -slideWidth : slideWidth;

          if (container.scrollBy) {
            container.scrollBy({ left: offset, behavior: 'smooth' });
          } else {
            container.scrollLeft += offset;
          }
        },

        onScroll: function(item, event) {
          var container = event.target;
          var index = Math.round(Math.abs(container.scrollLeft) / container.clientWidth);
          this.$set(item, 'activeImgIndex', index);
        }
      },
      mounted: function() {
        this.fetchProperties();
      }
    });
  </script>
